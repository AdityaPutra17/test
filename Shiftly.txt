    function formatRupiahText(angka) {
        return angka.toString().replace(/\B(?=(\d{3})+(?!\d))/g, ".");
    }

    function parseRupiah(string) {
        if (!string) return 0;
        return parseFloat(string.toString().replace(/\./g, '')) || 0;
    }

    function formatInputRupiah(input) {
        let value = input.value.replace(/[^,\d]/g, '').toString();
        let split = value.split(',');
        let sisa = split[0].length % 3;
        let rupiah = split[0].substr(0, sisa);
        let ribuan = split[0].substr(sisa).match(/\d{3}/gi);

        if (ribuan) {
            let separator = sisa ? '.' : '';
            rupiah += separator + ribuan.join('.');
        }

        input.value = rupiah;
    }

    let globalNetto = 0;

    function hitungPayroll() {
        let fields = [
            { id: 'gajiPokok', val: document.getElementById('gajiPokok').value.trim() },
            { id: 'hariLemburKerja', val: document.getElementById('hariLemburKerja').value.trim() },
            { id: 'jamLemburKerja', val: document.getElementById('jamLemburKerja').value.trim() },
            { id: 'hariLemburLibur', val: document.getElementById('hariLemburLibur').value.trim() },
            { id: 'jamLemburLibur', val: document.getElementById('jamLemburLibur').value.trim() },
            { id: 'insentifTambahan', val: document.getElementById('insentifTambahan').value.trim() },
            { id: 'bpjsPersen', val: document.getElementById('bpjsPersen').value.trim() },
            { id: 'pajakPersen', val: document.getElementById('pajakPersen').value.trim() },
            { id: 'potonganTambahan', val: document.getElementById('potonganTambahan').value.trim() }
        ];

        let hasError = false;
        fields.forEach(f => {
            let el = document.getElementById(f.id);
            if (f.val === "") {
                el.classList.add('error-input');
                hasError = true;
            } else {
                el.classList.remove('error-input');
            }
        });

        if (hasError) {
            alert(currentLang === 'id' ? "⚠️ Perhatian: Mohon isi semua kolom yang kosong." : "⚠️ Warning: Please fill in all empty fields.");
            return;
        }

        let gajiPokok = parseRupiah(fields[0].val);
        let hariLemburKerja = parseFloat(fields[1].val) || 0;
        let jamLemburKerja = parseFloat(fields[2].val) || 0;
        let hariLemburLibur = parseFloat(fields[3].val) || 0;
        let jamLemburLibur = parseFloat(fields[4].val) || 0;
        let insentifTambahan = parseRupiah(fields[5].val);
        let bpjsRate = (parseFloat(fields[6].val) || 0) / 100;
        let pajakRate = (parseFloat(fields[7].val) || 0) / 100;
        let potonganTambahan = parseRupiah(fields[8].val);

        let upahPerJam = gajiPokok / 173;
        let totalLemburKerja = 0;

        if (jamLemburKerja > 0) {
            let hari = hariLemburKerja > 0 ? hariLemburKerja : 1; 
            let jamPertama1_5 = Math.min(jamLemburKerja, hari); 
            let sisaJam2_0 = Math.max(0, jamLemburKerja - jamPertama1_5);

            totalLemburKerja = (jamPertama1_5 * 1.5 * upahPerJam) + (sisaJam2_0 * 2 * upahPerJam);
        }

        let totalLemburLibur = 0;
        if (jamLemburLibur > 0) {
            let hLibur = hariLemburLibur > 0 ? hariLemburLibur : 1;
            let totalJamLiburMaks8 = Math.min(jamLemburLibur, 8 * hLibur);
            let totalJamLiburLebih = Math.max(0, jamLemburLibur - totalJamLiburMaks8);
            
            totalLemburLibur = (totalJamLiburMaks8 * 2 * upahPerJam) + (totalJamLiburLebih * 3 * upahPerJam);
        }

        let totalLembur = totalLemburKerja + totalLemburLibur;
        let totalBruto = gajiPokok + totalLembur + insentifTambahan;

        let totalBpjs = totalBruto * bpjsRate;
        let pph21 = totalBruto * pajakRate;

        globalNetto = totalBruto - totalBpjs - pph21 - potonganTambahan;

        document.getElementById('resUpahJam').innerText = 'Rp ' + formatRupiahText(Math.round(upahPerJam));
        document.getElementById('resLembur').innerText = 'Rp ' + formatRupiahText(Math.round(totalLembur));
        document.getElementById('resInsentif').innerText = 'Rp ' + formatRupiahText(Math.round(insentifTambahan));
        document.getElementById('resTotalBpjs').innerText = '- Rp ' + formatRupiahText(Math.round(totalBpjs));
        document.getElementById('resPajak').innerText = '- Rp ' + formatRupiahText(Math.round(pph21));
        document.getElementById('resPotonganLain').innerText = '- Rp ' + formatRupiahText(Math.round(potonganTambahan));
        document.getElementById('resNetto').innerText = 'Rp ' + formatRupiahText(Math.round(globalNetto));

        document.getElementById('hasil').style.display = 'block';
    }

    function generateParticipantInputs(isRestoring = false) {
        let totalPerson = parseInt(document.getElementById('totalPerson').value) || 0;
        let container = document.getElementById('participantContainer');
        if(!isRestoring) container.innerHTML = '';

        if(totalPerson > 15) totalPerson = 15;

        let savedParticipants = JSON.parse(localStorage.getItem('shiftly_split_participants') || '[]');

        if(!isRestoring) {
            container.innerHTML = '';
            for(let i = 1; i <= totalPerson; i++) {
                let defaultName = `Teman ${i}`;
                let row = document.createElement('div');
                row.className = 'participant-row';
                row.innerHTML = `
                    <input type="checkbox" id="check_${i}" onchange="saveSplitParticipantsState()">
                    <input type="text" id="name_${i}" value="${defaultName}" style="flex:1;" oninput="saveSplitParticipantsState()">
                `;
                container.appendChild(row);
            }
            saveSplitParticipantsState();
        }
    }

    function saveSplitParticipantsState() {
        let totalPerson = parseInt(document.getElementById('totalPerson').value) || 0;
        let participantsData = [];

        for(let i = 1; i <= totalPerson; i++) {
            let chk = document.getElementById(`check_${i}`);
            let nameInput = document.getElementById(`name_${i}`);
            if(chk && nameInput) {
                participantsData.push({
                    name: nameInput.value,
                    checked: chk.checked
                });
            }
        }
        localStorage.setItem('shiftly_split_participants', JSON.stringify(participantsData));
        localStorage.setItem('shiftly_split_persons_count', totalPerson);
        updateSplitTemplateMessage();
    }

    function muatDataPatunganStorage() {
        let savedCount = localStorage.getItem('shiftly_split_persons_count');
        let savedParticipants = JSON.parse(localStorage.getItem('shiftly_split_participants') || '[]');
        let grandTotalSaved = localStorage.getItem('shiftly_split_grandtotal');
        let perPersonSaved = localStorage.getItem('shiftly_split_perperson');

        if(savedCount && savedParticipants.length > 0) {
            document.getElementById('totalPerson').value = savedCount;
            let container = document.getElementById('participantContainer');
            container.innerHTML = '';

            savedParticipants.forEach((p, idx) => {
                let i = idx + 1;
                let row = document.createElement('div');
                row.className = 'participant-row';
                row.innerHTML = `
                    <input type="checkbox" id="check_${i}" ${p.checked ? 'checked' : ''} onchange="saveSplitParticipantsState()">
                    <input type="text" id="name_${i}" value="${p.name}" style="flex:1;" oninput="saveSplitParticipantsState()">
                `;
                container.appendChild(row);
            });

            if(grandTotalSaved && perPersonSaved) {
                globalPerPersonShare = parseFloat(perPersonSaved);
                document.getElementById('resSplitGrandTotal').innerText = 'Rp ' + formatRupiahText(Math.round(parseFloat(grandTotalSaved)));
                document.getElementById('resSharePerPerson').innerText = 'Rp ' + formatRupiahText(Math.round(globalPerPersonShare));
                document.getElementById('hasilPatungan').style.display = 'block';
                updateSplitTemplateMessage();
            }
        }
    }

    let globalPerPersonShare = 0;

    function hitungPatungan() {
        let billEl = document.getElementById('totalBill');
        let personEl = document.getElementById('totalPerson');
        
        let billVal = billEl.value.trim();
        let personVal = personEl.value.trim();

        if(billVal === "" || personVal === "") {
            if(billVal === "") billEl.classList.add('error-input');
            if(personVal === "") personEl.classList.add('error-input');
            alert(currentLang === 'id' ? "⚠️ Perhatian: Mohon isi total subtotal tagihan dan jumlah orang." : "⚠️ Warning: Please fill in total bill and number of persons.");
            return;
        }
        billEl.classList.remove('error-input');
        personEl.classList.remove('error-input');

        let subtotalTagihan = parseRupiah(billVal);
        let taxRate = parseFloat(document.getElementById('splitTaxCategory').value) || 0;
        let jumlahOrang = parseInt(personVal) || 1;

        let totalPajak = subtotalTagihan * (taxRate / 100);
        let grandTotal = subtotalTagihan + totalPajak;
        globalPerPersonShare = grandTotal / jumlahOrang;

        localStorage.setItem('shiftly_split_grandtotal', grandTotal);
        localStorage.setItem('shiftly_split_perperson', globalPerPersonShare);

        document.getElementById('resSplitGrandTotal').innerText = 'Rp ' + formatRupiahText(Math.round(grandTotal));
        document.getElementById('resSharePerPerson').innerText = 'Rp ' + formatRupiahText(Math.round(globalPerPersonShare));
        
        if(document.getElementById('participantContainer').children.length === 0) {
            generateParticipantInputs(false);
        } else {

    function hitungTargetLembur() {
        let gapokEl = document.getElementById('targetGapok');
        let nominalEl = document.getElementById('targetNominal');
        let maksJamEl = document.getElementById('targetMaksJam');

        let gapokVal = gapokEl.value.trim();
        let nominalVal = nominalEl.value.trim();
        let maksJamVal = maksJamEl.value.trim();

        let hasError = false;
        if(gapokVal === "") { gapokEl.classList.add('error-input'); hasError = true; } else { gapokEl.classList.remove('error-input'); }
        if(nominalVal === "") { nominalEl.classList.add('error-input'); hasError = true; } else { nominalEl.classList.remove('error-input'); }
        if(maksJamVal === "") { maksJamEl.classList.add('error-input'); hasError = true; } else { maksJamEl.classList.remove('error-input'); }

        if(hasError) {
            alert(currentLang === 'id' ? "⚠️ Perhatian: Mohon lengkapi semua kolom target lembur." : "⚠️ Warning: Please fill in all overtime target fields.");
            return;
        }

        let gapok = parseRupiah(gapokVal);
        let targetNominal = parseRupiah(nominalVal);
        let maksJam1Hari = parseFloat(maksJamVal) || 1;

        let nominalRate = gapok / 173;
        let rateJam1 = nominalRate * 1.5;
        let rateJam2Plus = nominalRate * 2.0;

        let nominalMaksLemburHarian = 0;
        if(maksJam1Hari === 1) {
            nominalMaksLemburHarian = rateJam1 * 1;
        } else {
            nominalMaksLemburHarian = rateJam1 + (rateJam2Plus * (maksJam1Hari - 1));
        }

        let totalJamDibutuhkan = 0;
        if(targetNominal <= (rateJam1)) {
            totalJamDibutuhkan = targetNominal / rateJam1;
        } else {
            let sisaNominal = targetNominal - rateJam1;
            totalJamDibutuhkan = 1 + (sisaNominal / rateJam2Plus);
        }

        let estimasiHariKerja = totalJamDibutuhkan / maksJam1Hari;

        document.getElementById('resTargetUpahJam').innerText = 'Rp ' + formatRupiahText(Math.round(nominalRate));
        document.getElementById('resTargetRate1').innerText = 'Rp ' + formatRupiahText(Math.round(rateJam1));
        document.getElementById('resTargetRate2').innerText = 'Rp ' + formatRupiahText(Math.round(rateJam2Plus));
        document.getElementById('resTargetMaksHarian').innerText = 'Rp ' + formatRupiahText(Math.round(nominalMaksLemburHarian));
        document.getElementById('resTargetTotalJam').innerText = totalJamDibutuhkan.toFixed(1) + ' Jam';
        document.getElementById('resTargetHariDibutuhkan').innerText = Math.ceil(estimasiHariKerja) + ' Hari Kerja (± ' + totalJamDibutuhkan.toFixed(1) + ' jam total)';

        document.getElementById('hasilTarget').style.display = 'block';
    }

    function bandingkanSlip() {
        let slipEl = document.getElementById('gajiAsliSlip');
        let rawSlip = slipEl.value.trim();

        if (rawSlip === "") {
            slipEl.classList.add('error-input');
            alert(currentLang === 'id' ? "⚠️ Perhatian: Kolom slip gaji masih kosong!" : "⚠️ Warning: Payslip field is empty!");
            return;
        }
        slipEl.classList.remove('error-input');

        let slipAsli = parseRupiah(rawSlip);
        let container = document.getElementById('compareResultContainer');

        if(globalNetto === 0) {
            alert(currentLang === 'id' ? "Silakan hitung gaji Anda terlebih dahulu!" : "Please calculate your salary first!");
            return;
        }

        let selisih = slipAsli - globalNetto;
        let textSelisih = formatRupiahText(Math.abs(Math.round(selisih)));

        if(Math.abs(selisih) < 1000) {
            container.innerHTML = `<div class="compare-result match"><i class="fa-solid fa-circle-check"></i> ${currentLang === 'id' ? 'Sempurna! Hasil hitung cocok.' : 'Perfect! Calculation matches.'}</div>`;
        } else if(selisih > 0) {
            container.innerHTML = `<div class="compare-result match"><i class="fa-solid fa-circle-info"></i> ${currentLang === 'id' ? 'Slip lebih tinggi Rp ' + textSelisih : 'Payslip is higher by Rp ' + textSelisih}</div>`;
        } else {
            container.innerHTML = `<div class="compare-result diff"><i class="fa-solid fa-triangle-exclamation"></i> ${currentLang === 'id' ? 'Slip lebih rendah Rp ' + textSelisih : 'Payslip is lower by Rp ' + textSelisih}</div>`;
        }
    }

    let calcValue = "0";
    function calcAppend(val) {
        if(calcValue === "0" && !['+', '-', '*', '/'].includes(val)) {
            calcValue = val;
        } else {
            calcValue += val;
        }
        document.getElementById('calcDisplay').value = calcValue;
    }

    function calcClear() {
        calcValue = "0";
        document.getElementById('calcDisplay').value = calcValue;
    }

    function calcBack() {
        if(calcValue.length > 1) {
            calcValue = calcValue.slice(0, -1);
        } else {
            calcValue = "0";
        }
        document.getElementById('calcDisplay').value = calcValue;
    }

    function calcCompute() {
        try {
            let result = eval(calcValue.replace(/×/g, '*').replace(/÷/g, '/'));
            calcValue = String(Math.round(result * 100) / 100);
            document.getElementById('calcDisplay').value = calcValue;
        } catch (e) {
            document.getElementById('calcDisplay').value = "Error";
            calcValue = "0";
        }
    }

    function simpanTukarShift() {
        let tgl = document.getElementById('swapDate').value;
        let rekan = document.getElementById('swapColleague').value.trim();
        let ket = document.getElementById('swapNote').value.trim();

        if(!tgl || !rekan || !ket) {
            alert(currentLang === 'id' ? "⚠️ Mohon lengkapi semua data tukar shift!" : "⚠️ Please fill in all shift swap details!");
            return;
        }

        let daftarSwap = JSON.parse(localStorage.getItem('shiftly_swaps') || '[]');
        daftarSwap.push({ id: Date.now(), tgl, rekan, ket });
        localStorage.setItem('shiftly_swaps', JSON.stringify(daftarSwap));

        document.getElementById('swapDate').value = '';
        document.getElementById('swapColleague').value = '';
        document.getElementById('swapNote').value = '';

        muatDaftarShiftSwap();
        alert(currentLang === 'id' ? "✅ Catatan tukar shift berhasil disimpan!" : "✅ Shift swap note saved successfully!");
    }

    function muatDaftarShiftSwap() {
        let container = document.getElementById('swapListContainer');
        if(!container) return;

        let daftarSwap = JSON.parse(localStorage.getItem('shiftly_swaps') || '[]');
        if(daftarSwap.length === 0) {
            container.innerHTML = `<p style="font-size:12px; color: var(--text-muted);">${currentLang === 'id' ? 'Belum ada catatan tukar shift.' : 'No shift swap notes yet.'}</p>`;
            return;
        }

        let html = '';
        daftarSwap.forEach(item => {
            html += `
                <div class="swap-item">
                    <div class="swap-info">
                        <strong>📅 ${item.tgl}</strong> - ${item.rekan}<br>
                        <span style="color: var(--text-muted);">${item.ket}</span>
                    </div>
                    <button class="btn-delete-swap" onclick="hapusSwap(${item.id})"><i class="fa-solid fa-trash"></i></button>
                </div>
            `;
        });
        container.innerHTML = html;
    }

    function hapusSwap(id) {
        let daftarSwap = JSON.parse(localStorage.getItem('shiftly_swaps') || '[]');
        daftarSwap = daftarSwap.filter(item => item.id !== id);
        localStorage.setItem('shiftly_swaps', JSON.stringify(daftarSwap));
        muatDaftarShiftSwap();
    }
